async function generatePairingCode() { const number = document.getElementById('phoneNumber').value; const display = document.getElementById('pairCodeDisplay'); display.innerText = "GENERATING..."; try { // This calls the /pair route in your server.js const response = await fetch(`/pair?number=${number}`); const data = await response.json(); if (data.code) { display.innerText = data.code; // Shows the 8-digit code } else { display.innerText = "ERROR"; } } catch (err) { display.innerText = "OFFLINE"; } }
